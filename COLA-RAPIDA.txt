 1. SQL - CRIAR BANCO E TABELAS (com chave estrangeira)
------------------------------------------------------------------
DROP DATABASE IF EXISTS meu_banco;
CREATE DATABASE meu_banco CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE meu_banco;

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    login VARCHAR(50) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL
);

CREATE TABLE produtos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    preco DECIMAL(10,2) NOT NULL DEFAULT 0,
    quantidade INT NOT NULL DEFAULT 0,
    estoque_minimo INT NOT NULL DEFAULT 0
);

CREATE TABLE movimentacoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    produto_id INT NOT NULL,
    usuario_id INT NOT NULL,
    tipo ENUM('entrada','saida') NOT NULL,
    quantidade INT NOT NULL,
    data_movimentacao DATE NOT NULL,
    FOREIGN KEY (produto_id) REFERENCES produtos(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

 9. BUBBLE SORT (ordem alfabética das linhas do banco)
------------------------------------------------------------------
function bubbleSortPorCampo($lista, $campo) {
    $n = count($lista);
    for ($i = 0; $i < $n - 1; $i++) {
        for ($j = 0; $j < $n - 1 - $i; $j++) {
            if (strcasecmp($lista[$j][$campo], $lista[$j + 1][$campo]) > 0) {
                $temp = $lista[$j];
                $lista[$j] = $lista[$j + 1];
                $lista[$j + 1] = $temp;
            }
        }
    }
    return $lista;
}

 11. JOIN (mostrar nome do produto no histórico)
------------------------------------------------------------------
SELECT m.*, p.nome AS produto, u.nome AS usuario
FROM movimentacoes m
INNER JOIN produtos p ON m.produto_id = p.id
INNER JOIN usuarios u ON m.usuario_id = u.id
ORDER BY m.data_movimentacao DESC;
